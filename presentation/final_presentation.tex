\section{Motivation}
\begin{frame}{Motivation \& Problem Statement}
\begin{itemize}
  \item Data remains underused in organizations.
  \item Analytics tools require technical expertise.
  \item Manual reporting is slow and inconsistent.
  \item LLM insights need verification.
  \item Agentic AI enables automated analysis.
\end{itemize}
\end{frame}

\section{Introduction}
\begin{frame}{Introduction}
\begin{itemize}
  \item Web-based multi-agent data analyst.
  \item Targets financial and sales CSV data.
  \item Uses LangGraph for stateful orchestration.
  \item Combines statistics and LLM reasoning.
  \item Produces validated reports and charts.
\end{itemize}
\end{frame}

\section{Objectives}
\begin{frame}{Objectives}
\begin{itemize}
  \item Develop an autonomous multi-agent system that converts raw sales and financial data into professional analytical reports, trend summaries, and recommendations without manual coding
  \item Make basic financial analytics and sales optimization accessible to small and medium businesses through a web interface that does not require coding
\end{itemize}
\end{frame}

\section{Scope of Project}
\begin{frame}{Scope of Project}
\begin{itemize}
  \item Web upload supports CSV datasets.
  \item Internal loader supports more formats.
  \item Six-agent pipeline is implemented.
  \item FastAPI backend provides services.
  \item React frontend supports workflow.
\end{itemize}
\end{frame}

\section{Project Applications}
\begin{frame}{Project Applications}
\begin{itemize}
  \item Sales performance tracking.
  \item Financial reporting and budgeting.
  \item Retail and e-commerce analytics.
  \item Small business operations analysis.
  \item Reports for non-finance teams.
\end{itemize}
\end{frame}

  \item Precision@8 = 0.5938.

\section{Analysis / Discussion of Results}
\begin{frame}{Analysis / Discussion I}
\begin{itemize}
  \item Pipeline completed in about 52s.
  \item Standalone Qwen took 4m 12s.
  \item Pipeline was about 4.8x faster.
  \item Qwen had scaling errors.
  \item Validation blocked unchecked claims.
\end{itemize}
\end{frame}

\begin{frame}{Analysis / Discussion II}
\begin{itemize}
  \item Validation score was 100/100.
  \item Cohen's kappa was 1.0.
  \item Overall confidence was 0.97.
  \item Decision readiness was ready.
  \item Token metrics need instrumentation.
\end{itemize}
\end{frame}
